## ✅ 1. 冒泡排序（Bubble Sort）

**原理：** 相邻两个元素比较，大的往后换，直到整个数组有序。

```js
function bubbleSort(arr) {
  const len = arr.length;
    //外层控制i比较的轮数
  for (let i = 0; i < len - 1; i++) {
    let swapped = false;
      //里层控制每一轮比较的次数
    for (let j = 0; j < len - 1 - i; j++) {
      if (arr[j] > arr[j + 1]) {
        [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
        swapped = true;
      }
    }
    if (!swapped) break;
  }
  return arr;
}
```

* 最好时间复杂度：O(n)（已排序）
* 最坏/平均时间复杂度：O(n²)
* 空间复杂度：O(1)
* 稳定性：✅ 稳定

## 2.插入排序
```js

// 插入排序

let arr = [12,15,24,8,0,1,1,9];
function insertSort(arr) {
  const len = arr.length;
  // 从第二个数开始插，第一个默认有序
  for (let i = 1; i < len; i++) {
    // 当前要插入的数
    let current = arr[i];
    // 已排序区最后一个下标
    let j = i - 1;

    // 往前找：比current大的都往后挪一位
    while (j >= 0 && arr[j] > current) {
      arr[j + 1] = arr[j];
      j--;
    }
    // 放到空位
    arr[j + 1] = current;
  }
  return arr;
}
```
