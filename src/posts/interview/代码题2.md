 * @param {Array} arr 要扁平化的数组
 * @param {number} depth 扁平化深度，默认 1
 * @returns {Array} 扁平化后的数组

function flattenArray(arr, depth = 1) {
  // 入参校验
  if (!Array.isArray(arr)) {
    return [];
  }

  const result = [];

  for (const item of arr) {
    // 如果是数组 且 还能继续扁平化
    if (Array.isArray(item) && depth > 0) {
      result.push(...flattenArray(item, depth - 1));
    } else {
      result.push(item);
    }
  }

  return result;
}

function getTreeDepth(root) {
  if (!root) return 0;
  let queue = [root];
  let depth = 0;

  while (queue.length) {
    let size = queue.length;
    depth++;
    for (let i = 0; i < size; i++) {
      let node = queue.shift();
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
  }
  return depth;
}

function findTreeNode(treeList, targetId) {
  // 遍历当前层级
  for (const item of treeList) {
    if (item.id === targetId) {
      return item; // 找到直接返回
    }
    // 有子节点递归查找
    if (item.children && item.children.length) {
      const res = findTreeNode(item.children, targetId);
      if (res) return res;
    }
  }
  // 没找到
  return null;
}

function findNodePath(treeList, targetId, path = []) {
  for (const item of treeList) {
    const currPath = [...path, item.id];
    if (item.id === targetId) return currPath;
    if (item.children?.length) {
      const res = findNodePath(item.children, targetId, currPath);
      if (res) return res;
    }
  }
  return null;
}
function findNodeBFS(treeList, targetId) {
  // 空值保护
  if (!Array.isArray(treeList) || treeList.length === 0) return null;

  // 队列（标准 BFS 结构，不使用 shift()）
  const queue = [...treeList];

  while (queue.length > 0) {
    // 改用 pop() 从尾部取，O(1) 性能！
    const node = queue.pop();

    // 节点为空直接跳过
    if (!node) continue;

    // 找到目标
    if (node.id === targetId) return node;

    // children 安全判断 + 入队
    if (Array.isArray(node.children) && node.children.length > 0) {
      queue.push(...node.children);
    }
  }

  return null;
}

function compose(...funcs) {
  if (funcs.length === 0) {
    return arg => arg
  }

  if (funcs.length === 1) {
    return funcs[0]
  }

  return funcs.reduce(
    (a, b) =>
      (...args) =>
        a(b(...args))
  )
}

      if (treeMap[item.pid]) {
        treeMap[item.pid].children.push(treeMap[item.id]);
      }
    }
  });

  return result;
}

function treeToFlatIter(tree) {
  const res = []
  const queue = [...tree]

  let index = 0

  while (index < queue.length) {
    const node = queue[index++]

    const { children, ...item } = node

    res.push(item)

    if (children?.length) {
      queue.push(...children)
    }
  }

  return res
}

function treeToFlat(tree) {
  let res = []
  tree.forEach(item => {
    const { children, ...rest } = item
    res.push(rest)
    if (children && children.length) {
      res = res.concat(treeToFlat(children))
    }
  })
  return res
}
