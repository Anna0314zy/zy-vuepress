# 1. Text2Embedding_DashScopeHello.py（入门 Hello World）
```python
"""
阿里云 DashScope 文本向量化 Hello World
最简示例，测试 API 是否可用
"""
import os
from dashscope import TextEmbedding

# 配置 API Key
os.environ["DASHSCOPE_API_KEY"] = "你的阿里云 API Key"

def hello_embedding():
    # 最简单的向量化调用
    resp = TextEmbedding.call(
        model="text-embedding-v3",
        input="你好，这是一个测试文本"
    )

    if resp.status_code == 200:
        vec = resp.output["embeddings"][0]["embedding"]
        print("✅ 向量化成功！")
        print("向量维度：", len(vec))
        print("向量前 5 个值：", vec[:5])
    else:
        print("❌ 失败：", resp.code, resp.message)

if __name__ == "__main__":
    hello_embedding()
```

# 3. Text2Embedding_DashScope.py（基础标准版）
**最常用！业务直接用这个**
```python
"""
阿里云 DashScope 文本向量化基础版
支持：单条、批量、相似度计算
"""
import os
import numpy as np
from dashscope import TextEmbedding
from sklearn.metrics.pairwise import cosine_similarity

# ===================== 配置 =====================
os.environ["DASHSCOPE_API_KEY"] = "你的阿里云 API Key"
MODEL = "text-embedding-v3"
# =================================================

class Text2Embedding:
    def __init__(self):
        self.model = MODEL

    def encode(self, text: str):
        """单文本向量化"""
        resp = TextEmbedding.call(model=self.model, input=text)
        if resp.status_code == 200:
            return resp.output["embeddings"][0]["embedding"]
        raise Exception(f"请求失败：{resp.code}")

    def encode_batch(self, texts: list):
        """批量向量化"""
        resp = TextEmbedding.call(model=self.model, input=texts)
        if resp.status_code == 200:
            return [item["embedding"] for item in resp.output["embeddings"]]
        raise Exception(f"批量请求失败：{resp.code}")

    def similarity(self, vec1, vec2):
        """余弦相似度 0~1"""
        v1 = np.array(vec1).reshape(1, -1)
        v2 = np.array(vec2).reshape(1, -1)
        return cosine_similarity(v1, v2)[0][0]

# 测试
if __name__ == "__main__":
    emb = Text2Embedding()
    vec1 = emb.encode("我喜欢人工智能")
    vec2 = emb.encode("AI 技术很有趣")
    print("相似度：", emb.similarity(vec1, vec2))
```

# 4. Text2Embedding_DashScopePro.py（进阶企业版）
**包含：重试、多模态、RAG 检索、向量存储**
```python
"""
DashScope 进阶版 Pro
支持：
- 自动重试
- 多模态（文本+图像）
- RAG 知识库构建
- 语义检索
"""
import os
import time
import numpy as np
from dashscope import TextEmbedding, MultiModalEmbedding
from sklearn.metrics.pairwise import cosine_similarity

# ===================== 配置 =====================
os.environ["DASHSCOPE_API_KEY"] = "你的阿里云 API Key"
TEXT_MODEL = "text-embedding-v3"
MULTI_MODEL = "multimodal-embedding-v1"
MAX_RETRY = 3
# =================================================

class DashScopeEmbeddingPro:
    def __init__(self):
        self.knowledge_base = []  # 简易知识库

    def _retry(self, func, *args):
        for i in range(MAX_RETRY):
            try:
                return func(*args)
            except Exception as e:
                time.sleep(1)
        raise Exception("重试失败")

    def text_emb(self, text):
        return self._retry(self._text_emb, text)

    def _text_emb(self, text):
        resp = TextEmbedding.call(TEXT_MODEL, input=text)
        return resp.output["embeddings"][0]["embedding"]

    def add_knowledge(self, text):
        """添加到 RAG 知识库"""
        vec = self.text_emb(text)
        self.knowledge_base.append({"text": text, "vector": vec})

    def search(self, query, top_k=1):
        """语义检索（RAG 核心）"""
        q_vec = self.text_emb(query)
        scores = []
        for item in self.knowledge_base:
            sim = cosine_similarity(np.array(q_vec).reshape(1,-1),
                                    np.array(item["vector"]).reshape(1,-1))[0][0]
            scores.append((item["text"], sim))
        scores.sort(key=lambda x: x[1], reverse=True)
        return scores[:top_k]

# 测试 RAG 检索
if __name__ == "__main__":
    emb = DashScopeEmbeddingPro()
    emb.add_knowledge("Python 是一门解释型编程语言")
    emb.add_knowledge("人工智能可以模拟人类的思考方式")
    emb.add_knowledge("向量数据库用于存储高维向量")

    query = "什么是向量数据库？"
    result = emb.search(query)
    print("最匹配的知识：", result)
```
