### 💡 补充背景
这些脚本都围绕「Embedding（向量化）」这个核心功能：
- 把文本/图像等数据，转换成高维数字向量
- 是 RAG 知识库、语义搜索、文本相似度匹配、多模态检索的核心底层能力
- 国内场景优先用阿里云 DashScope（速度快、中文效果好），海外场景用 OpenAI Embedding

### 📌 给你一份 `DashScopePro.py` 的进阶版模板（适配多模态/批量/重试）
```python
import os
import time
import logging
from typing import List, Union
from dashscope import TextEmbedding
from dashscope import MultiModalEmbedding
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# ===================== 配置区 =====================
os.environ["DASHSCOPE_API_KEY"] = "你的阿里云API Key"
TEXT_EMBEDDING_MODEL = "text-embedding-v3"
MULTIMODAL_EMBEDDING_MODEL = "multimodal-embedding-v1"
MAX_RETRIES = 3  # 失败重试次数
# ===================================================


class DashScopeEmbeddingPro:
    """
    DashScope 进阶版 Embedding 工具类
    支持：文本/多模态向量化、批量处理、失败重试、相似度计算
    """
    def __init__(self):
        pass

    def _retry_call(self, call_func, *args, **kwargs):
        """带重试机制的API调用"""
        for attempt in range(MAX_RETRIES):
            try:
                response = call_func(*args, **kwargs)
                if response.status_code == 200:
                    return response
                logger.warning(f"API调用失败，尝试 {attempt+1}/{MAX_RETRIES}: {response.code}")
                time.sleep(1 * (attempt + 1))
            except Exception as e:
                logger.error(f"请求异常: {str(e)}")
                time.sleep(1 * (attempt + 1))
        raise Exception("API调用多次重试失败")

    def text_embedding(self, texts: Union[str, List[str]]) -> List[List[float]]:
        """
        文本向量化（支持单条/批量）
        """
        if isinstance(texts, str):
            texts = [texts]
        response = self._retry_call(
            TextEmbedding.call,
            model=TEXT_EMBEDDING_MODEL,
            input=texts
        )
        return [item["embedding"] for item in response.output["embeddings"]]

    def multimodal_embedding(self, text: str = None, image_path: str = None) -> List[float]:
        """
        多模态向量化（文本/图像/图文组合）
        """
        input = []
        if text:
            input.append({"text": text})
        if image_path:
            input.append({"image": image_path})

        response = self._retry_call(
            MultiModalEmbedding.call,
            model=MULTIMODAL_EMBEDDING_MODEL,
            input=input
        )
        return response.output["embeddings"][0]["embedding"]

    @staticmethod
    def cosine_similarity(vec1: List[float], vec2: List[float]) -> float:
        """计算余弦相似度"""
        vec1_np = np.array(vec1).reshape(1, -1)
        vec2_np = np.array(vec2).reshape(1, -1)
        return cosine_similarity(vec1_np, vec2_np)[0][0]


# 测试示例
if __name__ == "__main__":
    embedder = DashScopeEmbeddingPro()

    # 1. 单文本向量化
    vec1 = embedder.text_embedding("人工智能正在改变世界")[0]
    print(f"文本向量维度：{len(vec1)}")

    # 2. 批量文本向量化
    texts = ["大模型应用开发", "RAG知识库构建", "向量数据库实践"]
    vec_list = embedder.text_embedding(texts)
    print(f"批量生成向量数量：{len(vec_list)}")

    # 3. 多模态向量化（图文匹配）
    vec_image = embedder.multimodal_embedding(image_path="test.jpg")
    vec_text = embedder.multimodal_embedding(text="一张城市风景照片")
    print(f"图文相似度：{embedder.cosine_similarity(vec_image, vec_text):.4f}")
```
